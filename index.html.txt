        // Populate Tone Generator Selector
        function initToneGeneratorSelect() {

        // Helper to calculate exact note frequency based on A4 reference
        function calculateFrequency(noteIndex, octave, refA4 = 440) {
            const semitonesFromA4 = (noteIndex - 9) + (octave - 4) * 12;
            return refA4 * Math.pow(2, semitonesFromA4 / 12);
        }

        // Auto-correlation Algorithm for Pitch Detection with Noise Gate Threshold
        function autoCorrelate(buf, sampleRate) {
            let SIZE = buf.length;
            let rms = 0;

            for (let i = 0; i < SIZE; i++) {
                let val = buf[i];
                rms += val * val;
            }
            rms = Math.sqrt(rms / SIZE);

            // Noise Gate: Ignore ambient noise below 0.025 threshold
            if (rms < 0.025) return -1;

            let r1 = 0, r2 = SIZE - 1, thres = 0.2;
            for (let i = 0; i < SIZE / 2; i++) {
                if (Math.abs(buf[i]) < thres) {
                    r1 = i;
                    break;
                }
            }
            for (let i = 1; i < SIZE / 2; i++) {
                if (Math.abs(buf[SIZE - i]) < thres) {
                    r2 = SIZE - i;
                    break;
                }
            }

            buf = buf.slice(r1, r2);
            SIZE = buf.length;

            let c = new Array(SIZE).fill(0);
            for (let i = 0; i < SIZE; i++) {
                for (let j = 0; j < SIZE - i; j++) {
                    c[i] = c[i] + buf[j] * buf[j + i];
                }
            }

            let d = 0;
            while (c[d] > c[d + 1]) d++;
            let maxval = -1, maxpos = -1;
            for (let i = d; i < SIZE; i++) {
                if (c[i] > maxval) {
                    maxval = c[i];
                    maxpos = i;
                }
            }

            let T0 = maxpos;
            let x1 = c[T0 - 1], x2 = c[T0], x3 = c[T0 + 1];
            let a = (x1 + x3 - 2 * x2) / 2;
            let b = (x3 - x1) / 2;
            if (a) T0 = T0 - b / (2 * a);

            return sampleRate / T0;
        }

        // Convert Frequency to Musical Note Data
        function getNoteFromFrequency(freq) {
            const noteNum = 12 * (Math.log(freq / referenceA4) / Math.log(2));
            const noteIndex = Math.round(noteNum) + 69; // 69 is A4 MIDI note
            
            const octave = Math.floor(noteIndex / 12) - 1;
            const noteName = NOTE_NAMES_EN[noteIndex % 12];
            const noteNameAr = NOTE_NAMES_AR[noteIndex % 12];
            
            // Expected perfect frequency for this note
            const expectedFreq = referenceA4 * Math.pow(2, (noteIndex - 69) / 12);
            
            // Cents offset calculation (-50 to +50)
            const cents = Math.floor(1200 * Math.log2(freq / expectedFreq));

            return {
                name: noteName + octave,
                nameAr: noteNameAr,
                cents: cents,
                rawFreq: freq
            };
        }

        // Start Analyzing Pitch Loop
        function updatePitch() {
            const buffer = new Float32Array(2048);
            analyser.getFloatTimeDomainData(buffer);
            const pitch = autoCorrelate(buffer, audioCtx.sampleRate);

            if (pitch === -1 || pitch < 40 || pitch > 2000) {
                // Low signal / Quiet state - decay smoothed values slowly
                inTuneCounter = 0;
                isTuneLocked = false;
                lockIcon.classList.add('hidden');
                centerArcZone.setAttribute('opacity', '0.4');

                tuneStatusBadge.innerHTML = `<i class="fa-solid fa-waveform text-slate-500"></i> <span>اعزف أو غَنّ في الميكروفون...</span>`;
                tuneStatusBadge.className = "absolute bottom-2 px-4 py-1.5 rounded-full text-xs font-bold transition-all bg-slate-800 text-slate-400 border border-slate-700 flex items-center gap-1.5";
            } else {
                const rawNoteData = getNoteFromFrequency(pitch);

                // Apply Exponential Moving Average Smoothing
                if (smoothedFreq === 0) {
                    smoothedFreq = rawNoteData.rawFreq;
                    smoothedCents = rawNoteData.cents;
                } else {
                    smoothedFreq += (rawNoteData.rawFreq - smoothedFreq) * SMOOTHING_FACTOR;
                    smoothedCents += (rawNoteData.cents - smoothedCents) * SMOOTHING_FACTOR;
                }

                updateUI(rawNoteData.name, rawNoteData.nameAr, smoothedFreq, smoothedCents);
            }

            if (isListening) {
                requestAnimationFrame(updatePitch);
            }
        }

        // Short Beep Tone for Tuning Lock Feedback
        function triggerLockBeep() {
            if (!audioCtx) return;
            try {
                const osc = audioCtx.createOscillator();
                const g = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(880, audioCtx.currentTime); // A5 note
                g.gain.setValueAtTime(0.05, audioCtx.currentTime);
                g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
                osc.connect(g);
                g.connect(audioCtx.destination);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.15);
            } catch (e) {
                // Ignore audio autoplay restrictions if any
            }
        }

        // Update Gauge & UI Elements
        function updateUI(noteName, noteNameAr, freq, cents) {
            // Update Text Displays
            detectedNote.textContent = noteName;
            detectedNoteAr.textContent = noteNameAr;
            detectedFreq.textContent = freq.toFixed(1);
            
            const displayCents = Math.round(cents);
            centsValue.textContent = (displayCents > 0 ? `+${displayCents}` : `${displayCents}`);

            // Update Needle Rotation (-50 cents = -85deg, 0 cents = 0deg, +50 cents = +85deg)
            let clampedCents = Math.max(-50, Math.min(50, cents));
            let needleAngle = (clampedCents / 50) * 85; 
            needle.style.transform = `rotate(${needleAngle}deg)`;

            // Check Tuning Accuracy & Handle Lock State
            if (Math.abs(cents) <= 4) {
                inTuneCounter++;
                if (inTuneCounter >= STABILITY_THRESHOLD_FRAMES && !isTuneLocked) {
                    isTuneLocked = true;
                    triggerLockBeep();
                }
            } else {
                inTuneCounter = 0;
                isTuneLocked = false;
            }

            // Visual Colors & Status Feedback

        // Toggle Microphone Listening
        async function toggleMicrophone() {
            if (isListening) {
                // Stop Microphone
                if (micStream) {
                    micStream.getTracks().forEach(track => track.stop());
                }
                isListening = false;
                smoothedFreq = 0;
                smoothedCents = 0;
                micToggleBtn.className = "w-full py-4 px-6 rounded-2xl bg-gradient-to-r from-blue-600 to-emerald-500 hover:from-blue-500 hover:to-emerald-400 text-white font-bold text-lg shadow-xl transition-all duration-200 flex items-center justify-center gap-3";
                micBtnText.textContent = "تشغيل الميكروفون";
                micStatusHelp.textContent = "تم إيقاف الميكروفون. اضغط للبدء مجدداً.";
                return;
            }

            try {
                // Initialize Audio Context
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                micStream = await navigator.mediaDevices.getUserMedia({ audio: { echoCancellation: true, noiseSuppression: true } });
                
                const source = audioCtx.createMediaStreamSource(micStream);
                analyser = audioCtx.createAnalyser();
                analyser.fftSize = 4096;
                source.connect(analyser);

                isListening = true;
                micToggleBtn.className = "w-full py-4 px-6 rounded-2xl bg-gradient-to-r from-rose-600 to-red-500 hover:from-rose-500 hover:to-red-400 text-white font-bold text-lg shadow-xl transition-all duration-200 flex items-center justify-center gap-3";
                micBtnText.textContent = "إيقاف الميكروفون";
                micStatusHelp.textContent = "الميكروفون يعمل الآن. اعزف نغمتك ببطء لاستقرار المؤشر.";

                updatePitch();
            } catch (err) {
                alert("تعذر الوصول إلى الميكروفون. يرجى التأكد من منح الإذن في متصفحك.");
                console.error("Microphone Access Error:", err);
            }
        }

        // Switch Instrument Mode & Presets
        function setPresetMode(modeKey) {
            currentMode = modeKey;
            const preset = PRESETS[modeKey];
            
            // Update Mode Selector Active Buttons Styling
            document.querySelectorAll('.mode-btn').forEach(btn => {
                if (btn.dataset.mode === modeKey) {
                    btn.className = "mode-btn active-mode flex-1 min-w-[110px] py-2 px-3 rounded-xl font-semibold text-sm transition-all duration-200 flex items-center justify-center gap-2 bg-blue-600 text-white shadow-md";
                } else {
                    btn.className = "mode-btn flex-1 min-w-[110px] py-2 px-3 rounded-xl font-semibold text-sm transition-all duration-200 flex items-center justify-center gap-2 bg-slate-800 text-slate-300 hover:bg-slate-700";
                }
            });

            // Display Preset Strings Bar if applicable
            stringPresetBar.innerHTML = '';
            if (preset.notes && preset.notes.length > 0) {
                stringPresetBar.classList.remove('hidden');
                preset.notes.forEach(noteObj => {
                    const btn = document.createElement('button');
                    btn.className = "py-1.5 px-3 rounded-lg bg-slate-800/80 hover:bg-slate-700 border border-slate-700 text-xs font-bold text-slate-200 transition-all flex items-center gap-1.5";
                    btn.innerHTML = `<span class="text-blue-400 note-font">${noteObj.name}</span> <span>(${noteObj.label})</span>`;
                    btn.onclick = () => playReferenceTone(noteObj.freq);
                    stringPresetBar.appendChild(btn);
                });
            } else {
                stringPresetBar.classList.add('hidden');
            }
        }

        // Tone Generator Sound Player
        function playReferenceTone(freq) {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }

            if (isPlayingTone) {
                stopReferenceTone();
            }

            toneOscillator = audioCtx.createOscillator();
            const gainNode = audioCtx.createGain();

            toneOscillator.type = 'sine';
            toneOscillator.frequency.setValueAtTime(freq, audioCtx.currentTime);

            gainNode.gain.setValueAtTime(0.15, audioCtx.currentTime);
            
            toneOscillator.connect(gainNode);
            gainNode.connect(audioCtx.destination);

            toneOscillator.start();
            isPlayingTone = true;

            playToneBtn.className = "py-2 px-5 rounded-xl bg-amber-600 hover:bg-amber-500 text-white font-bold text-sm transition-all flex items-center justify-center gap-2";
            playToneIcon.className = "fa-solid fa-stop";
            playToneText.textContent = "إيقاف";
        }

        function stopReferenceTone() {
            if (toneOscillator) {
                toneOscillator.stop();
                toneOscillator.disconnect();
                toneOscillator = null;
            }
            isPlayingTone = false;

            playToneBtn.className = "py-2 px-5 rounded-xl bg-slate-800 hover:bg-slate-700 border border-slate-600 text-emerald-400 font-bold text-sm transition-all flex items-center justify-center gap-2";
            playToneIcon.className = "fa-solid fa-play";
            playToneText.textContent = "تشغيل";
        }
